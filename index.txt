    <script type="module">
        const btn = document.getElementById("MoreBtn");
        let skip = 7;
        const limit = 8;

        btn?.addEventListener("click", async (e) => {
            e.preventDefault();
            const grid = document.getElementById("pritelGrid");
            if (!grid) return;

            // Simple loading state
            btn.classList.add("opacity-50", "cursor-wait");
            
            try {
                const response = await fetch("/api/getEntries?" + new URLSearchParams({skip: skip.toString()}).toString());
                
                if (response.status === 200) {
                    const data = await response.json();
                    
                    if (data?.length) {
                        data.map((item, index) => {
                            // Create container div matching the new style
                            const card = document.createElement("div");
                            card.className = "group bg-white dark:bg-slate-800 rounded-2xl shadow-sm border border-slate-100 dark:border-slate-700 hover:shadow-lg hover:-translate-y-1 transition-all duration-300 animate-fade-in";
                            
                            card.innerHTML = `
                                <a href='/r/${item.slug}' class="block p-4">
                                    <div class="flex justify-between items-center mb-4 px-2">
                                        <span class="font-bold text-lg capitalize">${item.monthName}</span>
                                        <span class="text-slate-400 text-sm font-mono">${item.year}</span>
                                    </div>
                                    <div class="relative w-full aspect-[210/297] bg-slate-100 dark:bg-slate-700 rounded-lg overflow-hidden shadow-inner">
                                        <img src="${item.thumbnailUrl}" 
                                             width="${item.thumbnailWidth}" 
                                             height="${item.thumbnailHeight}" 
                                             alt="" 
                                             class="w-full h-full object-cover opacity-0 transition-opacity duration-500"
                                             onload="this.classList.remove('opacity-0')"
                                        />
                                    </div>
                                </a>
                            `;
                            grid.insertBefore(card, btn.parentElement); // Insert before button wrapper
                        });
                        skip += limit;
                    } else {
                        btn.innerHTML = "<span class='text-slate-400'>Žádná další čísla</span>";
                        btn.classList.add("pointer-events-none");
                    }
                }
            } catch (err) {
                console.error(err);
            } finally {
                btn.classList.remove("opacity-50", "cursor-wait");
            }
        });
    </script>
